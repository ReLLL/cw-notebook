cmake_minimum_required(VERSION 3.20)
project(cw_notebook VERSION 1.0.25 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(BUILD_PLUGIN "Build the SDR++Brown plugin" ON)
option(SANITIZE "Instrument standalone tests" OFF)
find_package(Threads REQUIRED)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  add_compile_options("-ffile-prefix-map=${CMAKE_SOURCE_DIR}=." "-ffile-prefix-map=$ENV{HOME}=~")
endif()
add_library(cw_core STATIC src/decoder.cpp src/journal.cpp src/engine.cpp)
target_include_directories(cw_core PUBLIC src)
target_link_libraries(cw_core PUBLIC Threads::Threads)
set_target_properties(cw_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_options(cw_core PRIVATE -Wall -Wextra -Wpedantic)
if(SANITIZE)
  target_compile_options(cw_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(cw_core PUBLIC -fsanitize=address,undefined)
endif()
enable_testing()
add_executable(cw_tests tests/tests.cpp)
target_link_libraries(cw_tests PRIVATE cw_core)
set(CW_RECORDING_FIXTURE "" CACHE FILEPATH "Optional private 8 kHz reception fixture")
if(CW_RECORDING_FIXTURE)
  target_compile_definitions(cw_tests PRIVATE CW_FIXTURE="${CW_RECORDING_FIXTURE}")
endif()
add_test(NAME cw_tests COMMAND cw_tests)
add_executable(cw_recovery_tests tests/recovery_tests.cpp)
target_link_libraries(cw_recovery_tests PRIVATE cw_core)
add_test(NAME cw_recovery_tests COMMAND cw_recovery_tests)
set(CW_IMGUI_SOURCE "" CACHE PATH "Optional matching host ImGui sources for transcript UI regression tests")
if(CW_IMGUI_SOURCE)
  add_executable(cw_transcript_tests tests/transcript_tests.cpp src/transcript_view.cpp
    "${CW_IMGUI_SOURCE}/imgui.cpp" "${CW_IMGUI_SOURCE}/imgui_draw.cpp"
    "${CW_IMGUI_SOURCE}/imgui_tables.cpp" "${CW_IMGUI_SOURCE}/imgui_widgets.cpp")
  target_include_directories(cw_transcript_tests PRIVATE src "${CW_IMGUI_SOURCE}")
  add_test(NAME cw_transcript_tests COMMAND cw_transcript_tests)
endif()
add_executable(cw_decode tools/decode.cpp)
target_link_libraries(cw_decode PRIVATE cw_core)
if(BUILD_PLUGIN)
  if(NOT APPLE)
    message(FATAL_ERROR "The host plugin currently supports macOS. Use BUILD_PLUGIN=OFF for portable core tests.")
  endif()
  set(BROWN_SOURCE "${CMAKE_SOURCE_DIR}/../SDRPlusPlusBrown" CACHE PATH "Brown source headers")
  set(BROWN_APP "/Applications/SDR++Brown.app" CACHE PATH "Installed Brown app")
  if(NOT EXISTS "${BROWN_SOURCE}/core/src/module.h" OR NOT EXISTS "${BROWN_APP}/Contents/Frameworks/libsdrpp_core.dylib")
    message(FATAL_ERROR "Set BROWN_SOURCE to matching Brown source headers and BROWN_APP to the installed SDR++Brown.app.")
  endif()
  add_library(cw_notebook MODULE src/plugin.cpp src/plugin_ui.cpp src/transcript_view.cpp)
  target_include_directories(cw_notebook SYSTEM PRIVATE "${BROWN_SOURCE}/core/src" "${BROWN_SOURCE}/core/src/imgui" /opt/homebrew/include)
  target_compile_definitions(cw_notebook PRIVATE CW_VERSION_STRING="${PROJECT_VERSION}" CW_VERSION_MAJOR=${PROJECT_VERSION_MAJOR} CW_VERSION_MINOR=${PROJECT_VERSION_MINOR} CW_VERSION_PATCH=${PROJECT_VERSION_PATCH})
  target_link_libraries(cw_notebook PRIVATE cw_core "${BROWN_APP}/Contents/Frameworks/libsdrpp_core.dylib")
  set_target_properties(cw_notebook PROPERTIES PREFIX "" SUFFIX ".dylib" BUILD_RPATH "${BROWN_APP}/Contents/Frameworks")
endif()
